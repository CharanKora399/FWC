\section*{1993 IMO Problems}

\textbf{First Day — July 18, 1993}

\begin{enumerate}

\item Let $f(x) = x^n + 5x^{n-1} + 3$, where $n > 1$ is an integer. Prove that $f(x)$ cannot be expressed as the product of two nonconstant polynomials with integer coefficients.

\item Let $D$ be a point inside acute triangle $ABC$ such that $\angle ADB = \angle ACB + \pi/2$ and $AC \cdot BD = AD \cdot BC$.
\begin{enumerate}
\item Calculate the ratio $\frac{AB \cdot CD}{AC \cdot BD}$.
\item Prove that the tangents at $C$ to the circumcircles of $\triangle ACD$ and $\triangle BCD$ are perpendicular.
\end{enumerate}

\item On an infinite chessboard, $n^2$ pieces are arranged in an $n \times n$ block. A move is a jump over an adjacent occupied square to an empty square beyond, removing the jumped piece. Find all $n$ for which the game can end with one piece.

\end{enumerate}

\textbf{Second Day — July 19, 1993}

\begin{enumerate}

\item For three points $P,Q,R$, define $m(PQR)$ as the minimum altitude length of $\triangle PQR$. Prove
$m(ABC) \le m(ABX) + m(AXC) + m(XBC)$.

\item Does there exist $f:\mathbb{N}\to\mathbb{N}$ such that  
$f(1)=2$,  
$f(f(n)) = f(n) + n$,  
$f(n) < f(n+1)$ for all $n$?

\item There are lamps $L_0,\dots,L_{n-1}$ in a circle. Step $s_i$: if $L_{i-1}$ is lit, switch $L_i$. Initially all on. Show:
\begin{enumerate}
\item There exists $M(n)$ after which all are on again.
\item If $n = 2^k$, then $M(n) = n^2 - 1$.
\item If $n = 2^k + 1$, then $M(n) = n^2 - n + 1$.
\end{enumerate}

\end{enumerate}

\section*{1994 IMO Problems}

\begin{enumerate}

\item Let $a_1,\dots,a_m$ be distinct elements of $\{1,\dots,n\}$ such that if $a_i + a_j \le n$ then $a_i+a_j = a_k$ for some $k$. Prove
$\frac{a_1+\cdots+a_m}{m} \ge \frac{n+1}{2}$.

\item $ABC$ is isosceles with $AB=AC$. $M$ midpoint of $BC$. $O$ on $AM$ with $OB \perp AB$. $Q$ on $BC$. $E$ on $AB$, $F$ on $AC$, with $E,Q,F$ collinear. Prove $OQ \perp EF$ iff $QE=QF$.

\item For integer $k$, let $f(k)$ be the count of numbers in $\{k+1,\dots,2k\}$ whose base-2 form has exactly three 1s.
\begin{enumerate}
\item For every $m$, there exists $k$ with $f(k)=m$.
\item Find all $m$ with exactly one such $k$.
\end{enumerate}

\item Find all $(m,n)$ such that $\frac{n^3+1}{mn-1}$ is an integer.

\item Find all $f:S\to S$ with $S=\{x>-1\}$ satisfying  
$f(x+f(y)+xf(y)) = y+f(x)+yf(x)$ and  
$\frac{f(x)}{x}$ strictly increasing on $(-1,0)$ and $(0,\infty)$.

\item Show there exists a set $A$ of positive integers such that for any infinite set of primes $S$, there exist $m\in A$ and $n\notin A$, each a product of $k\ge2$ distinct primes from $S$.

\end{enumerate}
