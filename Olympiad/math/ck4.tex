\section*{FIRST DAY}

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=*, itemsep=1em]

\item Find integers $a,b,c$ with $1<a<b<c$ such that $(a-1)(b-1)(c-1)$ divides $abc-1$.

\item Find all functions $f:\mathbb{R}\to\mathbb{R}$ such that $f(x^2+f(y)) = y + (f(x))^2$.

\item Nine points in space, no four coplanar, edges colored red/blue/uncolored. Find minimum colored edges guaranteeing a monochromatic triangle.

\item[]
\vspace{0.3cm}\hrule\vspace{0.2cm}
{\Large \textbf{SECOND DAY}}

\item Circle $C$, tangent line $L$, point $M$ on $L$. Find locus of $P$ such that $C$ is incircle of triangle $PQR$ where $M$ is midpoint of $QR$.

\item For finite set $S$ in space with projections $S_x,S_y,S_z$, prove $|S|^2 \le |S_x||S_y||S_z|$.

\item Define $S(n)$ = largest $k$ such that $n^2$ is sum of $k$ positive squares.
\begin{enumerate}[label=(\alph*)]
\item Prove $S(n)\le n^2-14$ for $n\ge4$.
\item Find $n$ with equality.
\item Show infinitely many such $n$ exist.
\end{enumerate}

\end{enumerate}

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=*, itemsep=1em]

\item Let $f(x)=x^n+5x^{n-1}+3$. Prove it is not product of two nonconstant integer polynomials.

\item Triangle geometry condition with point $D$.  
(a) Find $(AB\cdot CD)/(AC\cdot BD)$.  
(b) Prove certain tangents are perpendicular.

\item Chessboard jumping game with $n^2$ pieces. Determine $n$ for which one piece can remain.

\item[]
\vspace{0.3cm}\hrule\vspace{0.2cm}
{\Large \textbf{SECOND DAY}}

\item For points define $m(PQR)$ = minimum altitude. Prove $m(ABC)\le m(ABX)+m(AXC)+m(XBC)$.

\item Does there exist $f:\mathbb{N}\to\mathbb{N}$ with $f(1)=2$, $f(f(n))=f(n)+n$, strictly increasing?

\item Circular lamps switching process. Prove cycle length formulas for even/odd $n$.

\end{enumerate}

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=*, itemsep=1em]

\item Distinct $a_i$ in $\{1,\dots,n\}$ closed under sums $\le n$. Prove average $\ge (n+1)/2$.

\item Isosceles triangle configuration. Prove $OQ \perp EF$ iff $QE=QF$.

\item Let $f(k)$ count numbers in $\{k+1,\dots,2k\}$ with exactly three 1s in binary.
\begin{enumerate}[label=(\alph*)]
\item Show every $m$ occurs.
\item Find $m$ with unique $k$.
\end{enumerate}

\item Find all $(m,n)$ such that $(n^3+1)/(mn-1)$ is integer.

\item Find $f:S\to S$, $S>-1$, satisfying functional equation and monotonic ratio condition.

\item Show existence of integer set $A$ with prime-product property.

\end{enumerate}
